在开发 Mac 上先验证摄像头基本功能：

```bash
# 确认 USB 设备识别
system_profiler SPUSBDataType | grep -A 5 "HIKVISION\|U64"

# Python 验证
python3 -c "
import cv2
cap = cv2.VideoCapture(0)  # 或其他设备编号
if cap.isOpened():
    ret, frame = cap.read()
    print(f'分辨率: {frame.shape[1]}x{frame.shape[0]}')
    print(f'帧率测试...')
    import time
    start = time.time()
    for _ in range(100):
        cap.read()
    fps = 100 / (time.time() - start)
    print(f'实际帧率: {fps:.1f} FPS')
    cap.release()
else:
    print('摄像头打开失败')
"
```

### 2.3 采集脚本

```python
"""拍摄测试样本用于后续 AI 模型评估"""
import cv2
import os
from datetime import datetime

save_dir = "test_samples"
os.makedirs(save_dir, exist_ok=True)

cap = cv2.VideoCapture(0)
cap.set(cv2.CAP_PROP_FRAME_WIDTH, 1920)
cap.set(cv2.CAP_PROP_FRAME_HEIGHT, 1080)

print("按 's' 保存当前帧，按 'q' 退出")
while True:
    ret, frame = cap.read()
    if not ret:
        break
    cv2.imshow("Camera Test", frame)
    key = cv2.waitKey(1) & 0xFF
    if key == ord('s'):
        ts = datetime.now().strftime("%Y%m%d_%H%M%S")
        path = f"{save_dir}/sample_{ts}.jpg"
        cv2.imwrite(path, frame)
        print(f"已保存: {path}")
    elif key == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()
```

## Phase 3：AI 模型推理测试

### 3.1 YOLO 目标检测

```python
"""用采集的样本测试 YOLO 检测效果"""
from ultralytics import YOLO

model = YOLO("yolo11n.pt")  # 或自定义模型

results = model("test_samples/sample_xxx.jpg")
for r in results:
    print(f"检测到 {len(r.boxes)} 个目标")
    for box in r.boxes:
        print(f"  类别: {r.names[int(box.cls)]} 置信度: {box.conf:.2f}")
```

### 3.2 VLM 动作识别（Ollama）

```bash
# 确认 Ollama 和模型就绪
ollama list
curl http://localhost:11434/api/tags

# 测试推理
python3 -c "
import httpx, base64

with open('test_samples/sample_xxx.jpg', 'rb') as f:
    img_b64 = base64.b64encode(f.read()).decode()

resp = httpx.post('http://localhost:11434/api/generate', json={
    'model': 'qwen2.5vl:3b',
    'prompt': '这张图片中的操作员在做什么操作？请用中文简短描述。',
    'images': [img_b64],
    'stream': False,
}, timeout=120)

print(resp.json()['response'])
"
```

### 4.3 全链路验证流程

```
1. 管理员创建 SOP 模板（定义 3-5 个步骤）
2. 管理员创建工单（关联模板和工位）
3. 操作员在工位前按步骤操作
4. 系统自动识别并记录每个步骤
5. 故意做错一个步骤 → 触发报警
6. 确认报警 → 检查飞书是否收到通知
7. 完成所有步骤 → 检查完工检查结果
8. 查看视频回放
9. 导出工单数据为 Excel
```

## Phase 5：产线环境验证

### 5.1 环境因素

| 因素 | 测试内容 | 处理方法 |
|------|----------|----------|
| 工位照明 | 是否有阴影/反光 | 调整摄像头角度或补光 |
| 振动 | 产线振动是否影响拍摄 | 加固支架 |
| 遮挡 | 操作员手/身体遮挡 | 调整摄像头位置 |
| 工件反光 | 金属件反光导致识别失败 | 偏振滤镜或调整角度 |
| 网络 | 边缘端到服务端延迟 | ping 测试 < 10ms |
